  174:     status='queued'. Actual SMTP/Twilio/WhatsApp delivery
  175:     can be connected later.
  176:     """
  177: 
  178:     guardian = db.execute(
  179:         text(
  180:             """
  181:             SELECT
  182:                 guardian_name,
  183:                 guardian_mobile,
  184:                 alternate_mobile,
  185:                 notify_absence,
  186:                 notification_channel
  187:             FROM student_guardians
  188:             WHERE student_id = :student_id
  189:             LIMIT 1
  190:             """
  191:         ),
  192:         {
  193:             "student_id": student_id,
  194:         },
  195:     ).mappings().first()
  196: 
  197:     # No guardian record.
  198:     if not guardian:
  199:         return False
  200: 
  201:     # Guardian explicitly disabled absence alerts.
  202:     if not guardian["notify_absence"]:
  203:         return False
  204: 
  205:     channel = (
  206:         guardian["notification_channel"]
  207:         or "email"
  208:     ).strip().lower()
  209: 
  210:     mobile = (
  211:         guardian["guardian_mobile"]
  212:         or guardian["alternate_mobile"]
  213:     )
  214: 
  215:     subject = (
  216:         f"AttendX Absence Alert - {student_name}"
  217:     )
  218: 
  219:     message = (
  220:         f"Dear {guardian['guardian_name']},\n\n"
  221:         f"AttendX attendance alert for {student_name}.\n\n"
  222:         f"Attendance session: {session_title}\n"
  223:         f"Subject: {subject_name or 'Not specified'}\n"
  224:         f"Class: {class_name or 'Not specified'}\n\n"
  225:         f"The student was marked ABSENT after the attendance "
  226:         f"session was closed.\n\n"
  227:         f"This notification is queued by AttendX."
  228:     )
  229: 
  230:     # Prevent duplicate absence notification for the
  231:     # same student/session.
  232:     existing = db.execute(
  233:         text(
  234:             """
  235:             SELECT id
  236:             FROM notification_logs
  237:             WHERE student_id = :student_id
  238:               AND event_type = 'attendance_absent'
  239:               AND subject = :subject
  240:             LIMIT 1
  241:             """
  242:         ),
  243:         {
  244:             "student_id": student_id,
  245:             "subject": subject,
  246:         },
  247:     ).scalar()
  248: 
  249:     if existing:
  250:         return False
  251: 
  252:     # For SMS/WhatsApp, mobile is the correct recipient.
  253:     # For email, guardian email is not currently stored
  254:     # in student_guardians, so recipient remains NULL.
  255:     recipient = mobile if channel in {
  256:         "sms",
  257:         "whatsapp",
  258:         "phone",
  259:     } else None
  260: 
  261:     db.execute(
  262:         text(
  263:             """
  264:             INSERT INTO notification_logs
  265:             (
  266:                 student_id,
  267:                 event_type,
  268:                 channel,
  269:                 recipient,
  270:                 subject,
  271:                 message,
  272:                 status
  273:             )
  274:             VALUES
  275:             (
  276:                 :student_id,
  277:                 'attendance_absent',
  278:                 :channel,
  279:                 :recipient,
  280:                 :subject,
  281:                 :message,
  282:                 'queued'
  283:             )
  284:             """
  285:         ),
  286:         {
  287:             "student_id": student_id,
  288:             "channel": channel,
  289:             "recipient": recipient,
  290:             "subject": subject,
  291:             "message": message,
  292:         },
  293:     )
  294: 
  295:     return True

  298: def _finalize_absent_students(
  299:     db: Session,
  300:     session_row,
  301: ):
  302:     """
  303:     Phase 1 core logic.
  304: 
  305:     When a teacher closes a session:
  306: 
  307:     1. Load all active registered students.
  308:     2. Check attendance_events for this session.
  309:     3. Students without present/late attendance become ABSENT.
  310:     4. Create real attendance_events records.
  311:     5. Queue configured guardian absence notifications.
  312:     """
  313: 
  314:     session_id = session_row["id"]
  315: 
  316:     # Existing 70% attendance architecture does not yet have
  317:     # a student_class_assignments table. Therefore Phase 1
  318:     # uses all active registered students as the session roster.
  319:     #
  320:     # Class-specific student enrollment will be added in the
  321:     # Classes/Enrollment phase.
  322:     students = db.execute(
  323:         text(
  324:             """
  325:             SELECT
  326:                 s.id,
  327:                 s.student_code,
  328:                 s.name,
  329:                 s.email,
  330:                 s.roll_number
  331:             FROM students s
  332:             WHERE s.is_active = TRUE
  333:             ORDER BY
  334:                 CASE
  335:                     WHEN s.roll_number ~ '^[0-9]+$'
  336:                     THEN CAST(s.roll_number AS INTEGER)
  337:                     ELSE NULL
  338:                 END NULLS LAST,
  339:                 s.roll_number,
  340:                 s.name
  341:             """
  342:         )
  343:     ).mappings().all()
  344: 
  345:     if not students:
  346:         return {
  347:             "total_students": 0,
  348:             "present": 0,
  349:             "late": 0,
  350:             "absent": 0,
  351:             "notifications_queued": 0,
  352:         }
  353: 
  354:     # Get subject/class names for notification message.
  355:     session_info = db.execute(
  356:         text(
  357:             """
  358:             SELECT
  359:                 s.title,
  360:                 sub.name AS subject_name,
  361:                 c.name AS class_name,
  362:                 c.section
  363:             FROM attendance_sessions s
  364:             LEFT JOIN subjects sub
  365:                 ON sub.id = s.subject_id
  366:             LEFT JOIN classes c
  367:                 ON c.id = s.class_id
  368:             WHERE s.id = :session_id
  369:             """
  370:         ),
  371:         {
  372:             "session_id": session_id,
  373:         },
  374:     ).mappings().first()
  375: 
  376:     session_title = (
  377:         session_info["title"]
  378:         if session_info
  379:         else session_row["title"]
  380:     )
  381: 
  382:     subject_name = (
  383:         session_info["subject_name"]
  384:         if session_info
  385:         else None
  386:     )
  387: 
  388:     class_name = None
  389: 
  390:     if session_info:
  391:         class_name = session_info["class_name"]
  392: 
  393:         if session_info["section"]:
  394:             class_name = (
  395:                 f"{class_name} - "
  396:                 f"{session_info['section']}"
  397:             )
  398: 
  399:     present_count = 0
  400:     late_count = 0
  401:     absent_count = 0
  402:     notification_count = 0
  403: 
  404:     for student in students:
  405: 
  406:         existing = db.execute(
  407:             text(
  408:                 """
  409:                 SELECT
  410:                     id,
  411:                     status
  412:                 FROM attendance_events
  413:                 WHERE session_id = :session_id
  414:                   AND student_id = :student_id
  415:                 LIMIT 1
  416:                 """
  417:             ),
  418:             {
  419:                 "session_id": session_id,
  420:                 "student_id": student["id"],
  421:             },
  422:         ).mappings().first()
  423: 
  424:         if existing:
  425:             current_status = (
  426:                 existing["status"] or ""
  427:             ).lower()
  428: 
  429:             if current_status == "present":
  430:                 present_count += 1
  431: 
  432:             elif current_status == "late":
  433:                 late_count += 1
  434: 
  435:             elif current_status == "absent":
  436:                 absent_count += 1
  437: 
  438:             continue
  439: 
  440:         # Student never checked in.
  441:         db.execute(
  442:             text(
  443:                 """
  444:                 INSERT INTO attendance_events
  445:                 (
  446:                     session_id,
  447:                     student_id,
  448:                     method,
  449:                     status,
  450:                     verification_reason
  451:                 )
  452:                 VALUES
  453:                 (
  454:                     :session_id,
  455:                     :student_id,
  456:                     'session_close',
  457:                     'absent',
  458:                     'Automatically marked absent when teacher closed the session.'
  459:                 )
  460:                 """
  461:             ),
  462:             {
  463:                 "session_id": session_id,
  464:                 "student_id": student["id"],
  465:             },
  466:         )
  467: 
  468:         absent_count += 1
  469: 
  470:         queued = _queue_absence_notification(
  471:             db=db,
  472:             student_id=student["id"],
  473:             student_name=student["name"],
  474:             session_title=session_title,
  475:             subject_name=subject_name,
  476:             class_name=class_name,
  477:         )
  478: 
  479:         if queued:
  480:             notification_count += 1
  481: 
  482:     return {
  483:         "total_students": len(students),
  484:         "present": present_count,
  485:         "late": late_count,
  486:         "absent": absent_count,
  487:         "notifications_queued": notification_count,
  488:     }

   68:     lon1,
